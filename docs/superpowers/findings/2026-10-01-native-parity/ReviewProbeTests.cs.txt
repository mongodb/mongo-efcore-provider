    private static readonly string OutFile = Environment.GetEnvironmentVariable("REVIEW_PROBE_OUT") ?? "/tmp/review-probes.txt";
    private static readonly MongoQueryMode[] Modes = [MongoQueryMode.DriverLinq, MongoQueryMode.Native, MongoQueryMode.NativeOnly];

    private static void Record(string probe, Func<MongoQueryMode, object?> run)
    {
        var lines = new List<string> { $"### {probe}" };
        foreach (var mode in Modes)
        {
            string outcome;
            try
            {
                outcome = Format(run(mode));
            }
            catch (Exception ex)
            {
                var inner = ex;
                while (inner.InnerException != null && inner is System.Reflection.TargetInvocationException) inner = inner.InnerException;
                outcome = $"THROW {inner.GetType().Name}: {inner.Message.Split('\n')[0]}";
                if (outcome.Length > 260) outcome = outcome[..260];
            }
            lines.Add($"  {mode,-10} => {outcome}");
        }
        lock (OutFile) File.AppendAllLines(OutFile, lines);
    }

    private static string Format(object? value)
        => value switch
        {
            null => "<null>",
            string s => $"\"{s}\"",
            System.Collections.IEnumerable e => "[" + string.Join(", ", e.Cast<object?>().Select(Format)) + "]",
            _ => value.ToString() ?? "<null>"
        };

    private sealed class ProbeContext(DbContextOptions options, string prefix) : DbContext(options)
    {
        public DbSet<Row> Rows => Set<Row>();
        public DbSet<Customer> Customers => Set<Customer>();
        public DbSet<Address> Addresses => Set<Address>();
        public DbSet<Owner> Owners => Set<Owner>();
        public DbSet<Animal> Animals => Set<Animal>();
        public DbSet<Dog> Dogs => Set<Dog>();
        public DbSet<Shipment> Shipments => Set<Shipment>();
        public DbSet<Doc> Docs => Set<Doc>();
        public DbSet<Rate> Rates => Set<Rate>();
        public string Prefix { get; } = prefix;

        protected override void OnModelCreating(ModelBuilder mb)
        {
            mb.Entity<Row>().ToCollection(Prefix + "Rows");
            mb.Entity<Customer>(b =>
            {
                b.ToCollection(Prefix + "Customers");
                b.HasOne(c => c.Address).WithOne().HasForeignKey<Address>(a => a.CustomerId);
            });
            mb.Entity<Address>().ToCollection(Prefix + "Addresses");
            mb.Entity<Owner>().ToCollection(Prefix + "Owners");
            mb.Entity<Animal>().ToCollection(Prefix + "Animals");
            mb.Entity<Dog>();
            mb.Entity<Cat>();
            mb.Entity<Shipment>().ToCollection(Prefix + "Shipments");
            mb.Entity<Doc>(b =>
            {
                b.ToCollection(Prefix + "Docs");
                b.Property(x => x.Code).HasConversion(v => "p_" + v, v => v.Substring(2));
            });
            mb.Entity<Rate>().ToCollection(Prefix + "Rates");
        }
    }

    private sealed class PrefixCacheKeyFactory : IModelCacheKeyFactory
    {
        public object Create(DbContext context, bool designTime) => ((ProbeContext)context).Prefix;
    }

    private ProbeContext Ctx(string prefix, MongoQueryMode mode)
    {
        var b = new DbContextOptionsBuilder<ProbeContext>()
            .UseMongoDB(database.Client, database.MongoDatabase.DatabaseNamespace.DatabaseName)
            .ReplaceService<IModelCacheKeyFactory, PrefixCacheKeyFactory>()
            .ConfigureWarnings(x => x.Ignore(CoreEventId.ManyServiceProvidersCreatedWarning));
        new MongoDbContextOptionsBuilder(b).UseQueryMode(mode);
        return new ProbeContext(b.Options, prefix);
    }

    private IMongoCollection<BsonDocument> Raw(string prefix, string name)
        => database.MongoDatabase.GetCollection<BsonDocument>(prefix + name);

    private static List<string> RecentTitles(ProbeContext db)
        => db.Rows.AsNoTracking().Where(r => r.Created > DateTime.UtcNow.AddSeconds(-3)).Select(r => r.Title).ToList();

    private static List<string> RecentTitlesNow(ProbeContext db)
        => db.Rows.AsNoTracking().Where(r => r.Created > DateTime.Now.AddHours(-1)).Select(r => r.Title).ToList();

    [Fact]
    public void Probe_utcnow_add_staleness()
    {
        foreach (var mode in Modes)
        {
            var p = "P" + Guid.NewGuid().ToString("N")[..8];
            Raw(p, "Rows").InsertOne(new BsonDocument { { "_id", ObjectId.GenerateNewId() }, { "Title", "fresh" }, { "Created", DateTime.UtcNow } });
            string outcome;
            try
            {
                List<string> first, second;
                using (var db = Ctx(p, mode)) first = RecentTitles(db);
                Thread.Sleep(5000);
                using (var db = Ctx(p, mode)) second = RecentTitles(db);
                outcome = $"first={Format(first)} after5s={Format(second)} (expected [\"fresh\"] then [])";
            }
            catch (Exception ex) { outcome = $"THROW {ex.GetType().Name}: {ex.Message.Split('\n')[0]}"; }
            lock (OutFile) File.AppendAllLines(OutFile, [$"### UtcNow.AddSeconds(-3) staleness [{mode}]", "  " + outcome]);
        }

        Record($"Now.AddHours(-1) local-offset (TZ offset now {TimeZoneInfo.Local.GetUtcOffset(DateTime.Now)}) row Created=UtcNow-30min", m =>
        {
            var p = "P" + Guid.NewGuid().ToString("N")[..8];
            Raw(p, "Rows").InsertOne(new BsonDocument { { "_id", ObjectId.GenerateNewId() }, { "Title", "halfhour" }, { "Created", DateTime.UtcNow.AddMinutes(-30) } });
            using var db = Ctx(p, m);
            return RecentTitlesNow(db);
        });
    }
